% ---- helpers ---------------------------------------------------------------
\newcommand\fs[2]{\fontsize{#1}{#2}\selectfont}
% Full-page canvas in millimetres, origin bottom-left, page is 297 x 420.
\newenvironment{poster}
  {\begin{tikzpicture}[remember picture, overlay, x=1mm, y=1mm,
     shift={(current page.south west)}]}
  {\end{tikzpicture}}
% \qr{x}{y}{size}: the org's QR code (\QrPath, from common/qr/*.tex) in
% \CardInk, size mm square with its bottom-right corner at (x,y).
\newcommand\qr[3]{%
  \begin{scope}[shift={(#1-#3,#2+#3)}, x=#3mm/\QrModules, y=#3mm/\QrModules]
    \fill[\CardInk] \QrPath;
  \end{scope}}
